#+begin_src elisp
  (let ((use-package-expand-minimally t))
    (pp-emacs-lisp-code
     (macroexpand-all
      '(use-package foo
	 :config (+ 1 2)
	 :init (+ 3 4)
	 :config (+ 5 6)
	 :init (+ 7 9)))))
  =>
  (progn
    (+ 3 4) (+ 7 9)
    (require 'foo nil nil)
    (+ 1 2) (+ 5 6)
    t)
#+end_src
** 属性解释
`:config` 中的代码在包加载之后才能被加载
`:init` 在包前执行的代码
`:custom` 类似于通过 custom-set-variable 设置
** 找到配置项属于那个包
通过 `C-h f` 和 `C-h v` 查看函数和变量在那个包中，如果是c编写的就用 `use-packge emacs`吧
虽然可以统一放在`use-package simple`中，但会导致simple下的代码膨胀，还是分别放置吧

  ;; 不从包缓存中加载

  (push '(menu-bar-lines . 0) default-frame-alist)
  (push '(tool-bar-lines . 0) default-frame-alist)
  (push '(vertical-scroll-bars) default-frame-alist)

  ;; 禁止自动缩放窗口先
  (setq frame-inhibit-implied-resize t)

  (menu-bar-mode -1)
  (tool-bar-mode -1)
  (scroll-bar-mode -1)
  (setq inhibit-splash-screen t)
  (setq use-file-dialog nil)

  ;; 在这个阶段不编译
  (setq comp-deferred-compilation nil)
  (add-to-list 'default-frame-alist '(fullscreen . maximized))
  (provide 'early-init)
