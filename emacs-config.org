** package.el setting

用 MELPA 获取社区扩展包。package.el 是内置的，这里显式 :ensure nil 避免use-package 重复 require。

#+begin_src emacs-lisp

  (use-package package
    :ensure nil
    :config
    (add-to-list 'package-archives '("melpa" . "https://melpa.org/packages/"))
    (unless (bound-and-true-p package--initialized)
      (package-initialize)))


#+end_src
** default config
*** 核心编辑器行为
这些是 Emacs C 层的全局设置，不属于任何具体 .el 库，所以放进 use-package emacs 这个虚拟包中。

~bidi-display-reordering / bidi-paragraph-direction / bidi-inhibit-bpa~:
纯英文/代码工作流不需要双向文本渲染，关掉可以显著提高渲染性能，特别是在大文件滚动时减少 reorder 开销。
redisplay-skip-fontification-on-input: 输入时不阻塞 redisplay， 避免打字卡顿。
cursor-in-non-selected-windows nil: 非选中窗口不画光标，减少重绘。
fast-but-imprecise-scrolling: 滚动优先速度，牺牲精确 fontify，对日常浏览体验提升明显，需要精确 fontify 时可手动触发。
tab-width 4: 个人缩进习惯，兼顾多数语言的社区规范。
use-short-answers t: y/n 代替 yes/no，提高交互效率。
indent-tabs-mode nil: 强制用空格缩进，避免混用 tab/space 导致的对齐问题。
load-prefer-newer t: 当存在 .el 和 .elc 时优先加载较新者， 开发 Elisp 时不会因为残留的 .elc 而用旧代码。
read-process-output-max: 增大进程输出缓冲区到 4MB，LSP 等子进程通信量大时减少 read 系统调用次数。

#+begin_src elisp
  (use-package emacs
    :ensure nil
    :custom
    (bidi-display-reordering 'left-to-right)
    (bidi-paragraph-direction 'left-to-right)
    (bidi-inhibit-bpa t)
    (redisplay-skip-fontification-on-input t)
    (cursor-in-non-selected-windows nil)
    (fast-but-imprecise-scrolling t)
    (tab-width 4)
    (use-short-answers t)
    (indent-tabs-mode nil)
    (load-prefer-newer t)
    :config
    (setq read-process-output-max (* 4 1024 1024))
    (when (eq system-type 'darwin)
      (setq mac-option-modifier 'meta)
      (mac-auto-operator-composition-mode t)))

#+end_src

*** 编辑体验
 hl-line: 不启用全局模式（global-hl-line-mode -1），因为有些 major-mode 自己的高亮会与之冲突。只在进入 major-mode 后按需启用，
 且排除 vterm（终端模拟不需要行高亮，且会干扰终端输出渲染）。
 
 so-long: 打开超长行文件（如压缩后的 JS）时自动关闭耗时的 minor mode，避免 Emacs 卡死。对查看大日志文件也很关键。
 
 delsel: 选中文本后直接键入会替换选中内容，符合大多数编辑器的直觉。
 
 icomplete (fido-vertical-mode): 用垂直列表展示补全候选，取代默认的水平
 
 minibuffer 补全。Icomplete 比 Vertico/Consult 轻量，且 Emacs 30 内置，零额外依赖。
 
 display-line-numbers: 启用行号并用 relative 模式。relative 可以让组合命令（如 d5j）不需要心算行号差值，光标所在行仍显示绝对行号。

#+begin_src elisp
  (use-package hl-line
    :ensure nil
    :init
    (global-hl-line-mode -1)
    :preface
    (defun my/hl-line-activate ()
      (unless (derived-mode-p 'vterm-mode)
        (hl-line-mode 1)))
    :hook (after-change-major-mode . my/hl-line-activate))

  (use-package so-long
    :ensure nil
    :hook
    (after-init . global-so-long-mode))

  (use-package delsel
    :ensure nil
    :hook (after-init . delete-selection-mode))

  (use-package icomplete
    :ensure nil
    :hook (after-init . fido-vertical-mode))

  (use-package display-line-numbers
    :ensure nil
    :hook (after-init . global-display-line-numbers-mode)
    :custom
    (display-line-numbers-type 'relative))
#+end_src

*** 文件操作
auto-save-visited-mode: 在 focus-out 或 idle 时自动保存文件内容到磁盘，不是存到 #autosave# 临时文件。比传统 auto-save 更安全。
 
make-backup-files nil: 不生成 file~ 备份文件。个人不需要备份机制，版本控制已经承担了历史记录职责。
 
~auto-save-interval / auto-save-timeout~: 每 300 次按键或 30 秒空闲触发一次自动保存，平衡性能与数据安全。
 
executable-make-buffer-file-executable-if-script-p: 保存 .sh/.py 等脚本时自动加可执行权限，省去手动 chmod +x。

#+begin_src elisp
  (use-package files
    :ensure nil
    :hook
    (after-init . auto-save-visited-mode)
    (after-save . executable-make-buffer-file-executable-if-script-p)
    :custom
    (make-backup-files nil)
    (auto-save-default t)
    (auto-save-visited-interval 5)
    (auto-save-interval 300)
    (auto-save-timeout 30))
#+end_src

*** 重复命令
repeat-mode 让按完 C-c l n 后只需按 n/p 就能连续跳转 flymake 错误， 不需要每次都按 C-c l prefix。

eglot-error-repeat-map: 定义 n/p 分别对应 next/prev error，并用 put 挂到 flymake-goto-next-error 等命令的 repeat-map 属性上 这是 repeat-mode 识别"可重复命令"的标准做法。

#+begin_src elisp
  (use-package repeat
    :ensure nil
    :hook (after-init . repeat-mode)
    :config
    (defvar eglot-error-repeat-map
      (let ((map (make-sparse-keymap)))
        (define-key map "n" #'flymake-goto-next-error)
        (define-key map "p" #'flymake-goto-prev-error)
        map)
      "Keymap to repeat flymake error navigation.")
    (put #'flymake-goto-next-error 'repeat-map 'eglot-error-repeat-map)
    (put #'flymake-goto-prev-error 'repeat-map 'eglot-error-repeat-map))
#+end_src

*** 自动刷新
 文件在 Emacs 外被修改后自动重载缓冲区，配合 version control 非常有用（git checkout、git pull 后不用手动 revert-buffer）。
 
 global-auto-revert-non-file-buffers t: 也刷新 Dired、Buffer Menu 等非文件缓冲区，保持文件管理视图同步。
 
 auto-revert-interval 3: 每 3 秒检查一次，比默认值更及时。
 
 auto-revert-remote-files nil: 远程文件（Tramp）不自动刷新，避免网络 IO。
 
 auto-revert-use-notify t: 优先用 inotify/kqueue 监听，不用轮询。
 
 auto-revert-avoid-polling nil: 当 notify 不可用时允许回退到轮询。
 
 auto-revert-verbose t: 发生自动重载时显示消息，便于察觉文件变更。
 
#+begin_src elisp
  (use-package autorevert
    :ensure nil
    :commands (auto-revert-mode global-auto-revert-mode)
    :hook
    (after-init . global-auto-revert-mode)
    :custom
    (global-auto-revert-non-file-buffers t)
    (auto-revert-interval 3)
    (auto-revert-remote-files nil)
    (auto-revert-use-notify t)
    (auto-revert-avoid-polling nil)
    (auto-revert-verbose t))
#+end_src

*** 窗口管理
switch-to-buffer-in-dedicated-window 'pop: 在专用窗口中不能用 switch-to-buffer 切换 buffer，避免意外覆盖专用窗口内容。
 
switch-to-buffer-obey-display-actions t: 让 switch-to-buffer 也遵守display-buffer-alist 规则，统一窗口行为。

scroll-conservatively 101 / scroll-margin 0 / scroll-preserve-screen-position t: 光标到达窗口边界时只滚动最少行数，不会"跳跃式"滚动，保持阅读连续性。

auto-window-vscroll nil: 禁止垂直微滚动，避免行部分可见时的渲染问题。

display-buffer-alist:
   *Dictionary* 用侧边栏（宽度 70），不用弹窗，方便边看边输。
   *Help* 优先重用已有窗口，没有才弹新窗口，减少窗口数量。
   
#+begin_src elisp
  (use-package window
    :ensure nil
    :custom
    (switch-to-buffer-in-dedicated-window 'pop)
    (switch-to-buffer-obey-display-actions t)
    (scroll-conservatively 101)
    (scroll-margin 0)
    (scroll-preserve-screen-position t)
    :config
    (setq auto-window-vscroll nil)
    (add-to-list 'display-buffer-alist
                 '("^\\*Dictionary\\*"
                   (display-buffer-in-side-window)
                   (side . left)
                   (window-width . 70)))
    (add-to-list 'display-buffer-alist
                 '("\\*Help\\*"
                   (display-buffer-reuse-window display-buffer-pop-up-window)))
    )
#+end_src

*** 会话持久化
recentf: 记录最近打开的文件，C-x C-r 快速打开。排除压缩文件/图片/autoload 等无关文件，减少列表噪音。

recentf-auto-cleanup: daemon 模式下每 300 秒自动清理不存在的文件，非 daemon 模式则在退出时清理。

savehist: 保存 minibuffer 历史、kill-ring、mark-ring、搜索历史到文件，下次启动恢复，减少重复输入。

saveplace: 记住每个文件上次光标位置，重启后回到离开时的位置。

save-place-limit 400: 最多记住 400 个文件的位置。

#+begin_src elisp

** programming
*** LSP 语言服务
Eglot 是 Emacs 内置的 LSP 客户端，比 LSP-mode 更轻量、与 Emacs内置功能集成更好（flymake、xref、eldoc 等）。

eglot-ensure: 进入 prog-mode / latex-mode 时自动启动 LSP。

按键设计：
   C-c l prefix 是所有 LSP 操作的入口，l 取 language 之意。
   organize-imports / format / code-actions 是高频重构操作。
   eldoc / find-implementation / find-declaration / rename 是日常
   代码导航操作。
   reconnect / shutdown / events-buffer / stderr-buffer 用于调试
   和管理 LSP 服务器。
   flymake-goto-prev/next-error 放在这里是因为 diagnostics 是
   Eglot 的核心输出，放在相同 keymap 下更顺手。

 服务器配置：
   Python: rass (Rust 写的 LSP 客户端) + basedruff (Ruff 的 LSP 包装)。
   JavaScript: rass → typescript-language-server → biome lsp-proxy。
   HTML: biome 直接做 LSP。
   TeX: texlab 配合 tectonic 编译。

 texlab workspace 配置：
   build: 用 tectonic 编译，:onSave t 保存即编译，适合 LaTeX 即时预览。
   forwardSearch: 用 emacsclient 打开 PDF 并定位到对应源行。
   chktex: 保存和打开时自动检查 LaTeX 语法。
#+begin_src elisp
  (use-package eglot
    :ensure nil
    :hook
    ((prog-mode . eglot-ensure)
     (latex-mode . eglot-ensure))
    :bind
    (:map eglot-mode-map
          ("C-c l l" . eglot-code-action-organize-imports)
          ("C-c l f" . eglot-format)
          ("C-c l o" . eglot-code-action-organize-imports)

          ;; --- navigation & documentation ---
          ("C-c l h" . eldoc)
          ("C-c l i" . eglot-find-implementation)
          ("C-c l t" . eglot-find-declaration)
          ("C-c l r" . eglot-rename)
          ("C-c l a" . eglot-code-actions)

          ;; --- server management ---
          ("C-c l s" . eglot-reconnect)
          ("C-c l q" . eglot-shutdown)
          ("C-c l e" . eglot-events-buffer)
          ("C-c l w" . eglot-stderr-buffer)
          ("C-c l p" . flymake-goto-prev-error)
          ("C-c l n" . flymake-goto-next-error)
          )
    :config
    (add-to-list 'eglot-server-programs
                 '(python-base-mode . ("rass" "basedruff")))
    (add-to-list 'eglot-server-programs
                 '(js-base-mode . ("rass" "--" "typescript-language-server" "--stdio" "--" "biome" "lsp-proxy")))
    (add-to-list 'eglot-server-programs
                 '(html-ts-mode . ("biome" "lsp-proxy")))
    (setq-default eglot-workspace-configuration
                  '(:texlab (:build (:executable "tectonic"
                                                 :args ["-X" "compile" "-Z" "shell-escape" "%f"]
                                                 :onSave t)
                                    :forwardSearch (:executable "emacsclient"

*** Tree-sitter 语法解析引擎
 treesit-font-lock-level 4: 最高等级的语法高亮，使用完整的 Tree-sitter查询捕获信息，比传统 regex font-lock 更准确。

mp-setup-install-grammars: 自动安装缺失的 Tree-sitter grammar。Emacs 30+ 内置的 grammar 列表不完整，手动定义常用语言。

      (interactive)
      (dolist (grammar
               '(
                 (bash . ("https://github.com/tree-sitter/tree-sitter-bash"))
                 (c .  ("https://github.com/tree-sitter/tree-sitter-c"))
                 (cpp . ("https://github.com/tree-sitter/tree-sitter-cpp"))
                 (css . ("https://github.com/tree-sitter/tree-sitter-css"))
                 (go . ("https://github.com/tree-sitter/tree-sitter-go"))

                 (javascript . ("https://github.com/tree-sitter/tree-sitter-javascript"))
                 (json . ("https://github.com/tree-sitter/tree-sitter-json"))
                 (python . ("https://github.com/tree-sitter/tree-sitter-python"))
                 (rust . ("https://github.com/tree-sitter/tree-sitter-rust"))
                 (toml . ("https://github.com/ikatyang/tree-sitter-toml"))
                 ))
        (add-to-list 'treesit-language-source-alist grammar)
        (unless (treesit-language-available-p (car grammar))
          (treesit-install-language-grammar (car grammar)))))
    :config
